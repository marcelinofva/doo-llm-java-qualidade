<design>
1) Domain Entities (Classes):
   - User
   - Sponsor
   - Payment
   - Sponsorship
   - JobPosting

2) Single Responsibility of Each Class:
   - User: Manages user information and authentication.
   - Sponsor: Manages sponsor information and sponsorship details.
   - Payment: Handles payment processing and transaction details.
   - Sponsorship: Manages sponsorship levels and associated payments.
   - JobPosting: Manages job postings and sponsorship links.

3) Relationships:
   - User has many Sponsorships
   - Sponsor has many Sponsorships
   - Sponsorship has one Payment
   - Sponsorship has one JobPosting
   - User has many JobPostings

4) Public Methods:
   - User: login(), register(), identify()
   - Sponsor: createSponsorship(), linkSponsorshipToJob()
   - Payment: processPayment()
   - Sponsorship: getDetails(), getPaymentStatus()
   - JobPosting: postJob(), getDetails()
</design>

public class Sponsor {
    private String name;
    private String contactInfo;
    private List<Sponsorship> sponsorships;

    public Sponsor(String name, String contactInfo) {
        this.name = name;
        this.contactInfo = contactInfo;
        this.sponsorships = new ArrayList<>();
    }

    public void createSponsorship(Sponsorship sponsorship) {
        sponsorships.add(sponsorship);
    }

        sponsorship.setJobPosting(jobPosting);
    }

    public List<Sponsorship> getSponsorships() {
        return sponsorships;
    }
}

public class Payment {
    private double amount;
    private String transactionId;

    public Payment(double amount) {
        this.amount = amount;
        this.transactionId = generateTransactionId();
    }

    public String processPayment() {
        // Payment processing logic
        return transactionId;
    }

    private String generateTransactionId() {
        // Generate a unique transaction ID

    }

    public double getAmount() {
        return amount;
    }

    public String getTransactionId() {

public class Sponsorship {
    private double amount;
    private String level;
    private Payment payment;
    private JobPosting jobPosting;

    public Sponsorship(double amount, String level) {
        this.amount = amount;
        this.level = level;
        this.payment = null;
        this.jobPosting = null;
    }

    public void setPayment(Payment payment) {
        this.payment = payment;
    }

    public void setJobPosting(JobPosting jobPosting) {
        this.jobPosting = jobPosting;
    }

    public double getAmount() {
        return amount;
    }

    public String getLevel() {
        return level;
    }

    public Payment getPayment() {
        return payment;
    }

    public JobPosting getJobPosting() {
        return jobPosting;
    }

    public String getDetails() {
        return "Sponsorship Level: " + level + ", Amount: $" + amount;
    }

    public String getPaymentStatus() {
        if (payment == null) {
            return "Payment Pending";
        } else {
            return "Payment Processed: " + payment.getTransactionId();
        }
    }
}

public class Main {
    public static void main(String[] args) {
        User user = new User("admin", "password");
        if (user.login("admin", "password")) {
            System.out.println("User logged in successfully.");
        } else {
            System.out.println("Invalid credentials.");
        }

        Sponsor sponsor = new Sponsor("Tech Innovations", "contact@techinnovations.com");
        Sponsorship sponsorship = new Sponsorship(350, "Company Sponsorship");
        Payment payment = new Payment(350);
        sponsorship.setPayment(payment);
        sponsor.createSponsorship(sponsorship);

        JobPosting jobPosting = new JobPosting("Software Engineer", "Looking for a talented software engineer.");
        sponsorship.setJobPosting(jobPosting);
        sponsor.linkSponsorshipToJob(sponsorship, jobPosting);

        System.out.println(sponsorship.getDetails());
        System.out.println(sponsorship.getPaymentStatus());
        System.out.println(jobPosting.getDetails());
    }
}
