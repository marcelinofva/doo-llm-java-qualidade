// File: Main.java
public class Main {
    public static void main(String[] args) {
        PlanReviewStaff staff = new PlanReviewStaff();
        PlanReviewStaffSupervisor supervisor = new PlanReviewStaffSupervisor();
        Applicant applicant = new Applicant();
        Plan plan = new Plan();
        CodeModification codeModification = new CodeModification();
        Bond bond = new Bond();
        InspectionSupervisor inspectionSupervisor = new InspectionSupervisor();

        staff.assignPlanForReview(plan);
        supervisor.managePlanReviewerWorkload(staff);
        applicant.requestPlanReviewMeeting(staff);
        staff.conductPlanReviewMeeting(applicant, plan);
        staff.reviewPlans(plan);
        staff.reviewCodeModifications(codeModification);
        supervisor.manageBonds(bond);
        inspectionSupervisor.manageInspectorWorkload();
    }
}

// File: Plan.java
public class Plan {
    private String id;
    private String details;

    public Plan(String id, String details) {
        this.id = id;
        this.details = details;
    }

    public String getId() {
        return id;
    }

    public String getDetails() {
        return details;
    }
}

// File: PlanReviewStaff.java
public class PlanReviewStaff {
    public void assignPlanForReview(Plan plan) {
        System.out.println("Plan assigned for review: " + plan.getId());
    }

    public void conductPlanReviewMeeting(Applicant applicant, Plan plan) {
        System.out.println("Conducted plan review meeting for applicant: " + applicant.getName() + " and plan: " + plan.getId());
    }

    public void reviewPlans(Plan plan) {
        System.out.println("Reviewed plans for compliance: " + plan.getId());
    }

    public void reviewCodeModifications(CodeModification codeModification) {
        System.out.println("Reviewed code modifications: " + codeModification.getId());
    }
}
