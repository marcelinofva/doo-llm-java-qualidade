// File: MemberProfile.java
public class MemberProfile {
    private String name;
    private String role;
    private String experience;
    private String upcomingClasses;
    private String email;
    private boolean isPrivate;
    private boolean isEmailPrivate;

    public MemberProfile(String name, String role, String experience, String upcomingClasses, String email, boolean isPrivate, boolean isEmailPrivate) {
        this.name = name;
        this.role = role;
        this.experience = experience;
        this.upcomingClasses = upcomingClasses;
        this.email = email;
        this.isPrivate = isPrivate;
        this.isEmailPrivate = isEmailPrivate;
    }

    public String getName() {
        return name;
    }

    public void setName(String name) {
        this.name = name;
    }

    public String getRole() {
        return role;
    }

    public void setRole(String role) {
        this.role = role;
    }

    public String getExperience() {
        return experience;
    }

    public void setExperience(String experience) {
        this.experience = experience;
    }

    public String getUpcomingClasses() {
        return upcomingClasses;
    }

    public void setUpcomingClasses(String upcomingClasses) {
        this.upcomingClasses = upcomingClasses;
    }

    public String getEmail() {
        return isPrivate ? "Private" : email;
    }

    public void setEmail(String email) {
        this.email = email;
    }

    public boolean isPrivate() {
        return isPrivate;
    }

    public void setPrivate(boolean private) {
        isPrivate = private;
    }

    public boolean isEmailPrivate() {
        return isEmailPrivate;
    }

    public void setEmailPrivate(boolean emailPrivate) {
        isEmailPrivate = emailPrivate;
    }
}

// File: Main.java
public class Main {
    public static void main(String[] args) {
        MemberProfile memberProfile = new MemberProfile("John Doe", "Practitioner", "10 years of experience in Scrum", "Upcoming Classes: Advanced Scrum Master, Scrum for Teams", "john.doe@example.com", true, false);
        System.out.println("Name: " + memberProfile.getName());
        System.out.println("Role: " + memberProfile.getRole());
        System.out.println("Experience: " + memberProfile.getExperience());
        System.out.println("Upcoming Classes: " + memberProfile.getUpcomingClasses());
        System.out.println("Email: " + memberProfile.getEmail());
        System.out.println("Is Profile Private: " + memberProfile.isPrivate());
        System.out.println("Is Email Private: " + memberProfile.isEmailPrivate());
    }
}
